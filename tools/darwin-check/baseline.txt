# Packages of the module that do not build for darwin yet.
#
# Windows is still the reference platform, so these packages keep their
# Windows-only implementation until the macOS port lands. tools/darwin-check
# builds every package for darwin without cgo and requires this list to match
# exactly: a new failure is a regression, and a package that builds again must
# be removed from this file.
#
# One package per line; # starts a comment.
cmd/tuick-tack          # wires the Windows-only daemon and TUI bridges
internal/browser        # ShellExecuteW through golang.org/x/sys/windows
internal/daemon         # sound, console, autostart, launch and window bridges are Windows-only
internal/explorer       # explorer_windows.go only (explorer /select)
internal/install        # reads the Inno uninstall entry from the registry
internal/instance       # named mutexes in instance_windows.go only
internal/ipc            # go-winio named pipes through golang.org/x/sys/windows
internal/termwin        # console window lookup in termwin_windows.go only
internal/tray           # fyne.io/systray needs cgo on darwin; the icons are ICO
internal/tui            # imports the Windows-only browser, explorer, install and termwin bridges
internal/updater        # updater_windows.go only
internal/winproc        # CreateProcess helpers in winproc_windows.go only
internal/wt             # Windows Terminal profile lookup in wt_windows.go only
